```bash
  -d '{
    "model": "glm-5.3-flash",
```

```python

client = OpenAI(api_key="sk-你的中转Key", base_url="https://catrouter.net/v1")

    model="glm-5.3-flash",
    messages=[{"role": "user", "content": "写一段产品介绍"}],
    stream=True,
)
for chunk in resp:
    print(chunk.choices[0].delta.content or "", end="")
```
